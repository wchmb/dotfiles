# <type>[<scope>][!]: <subject>
# |<----          use max 50 chars           ---->|

# [body]
# |<----             wrap body at 72 characters                    ---->|

# [footer(s)]

# --- COMMIT END ---
# <type>    can be:
# - feat     (new feature)
# - fix      (bug fix)
# - perf     (performance improvement)
# - refactor (clean code)
# - style    (formatting)
# - test     (tests)
# - docs     (documentation)
# - chore    (grunt tasks: tooling, housekeeping or setup)
# - bump     (version bumps: release, tags or dependencies)
# - cicd     (CI/CD)
# - revert   (rollback)
# - merge    (merge commit)
# - wip      (work in progress; avoid on main/release branches)
#
# <scope>   describes the area affected, within parenthesis, e.g. (admin)
# <subject> uses the imperative mood, is lowercase, and has no period.
# !         marks a breaking change.
#
# Body rules:
# - Separate subject from body with one blank line
# - Explain what changed and why, not how
# - Wrap lines at 72 characters
# - Use "-" for bullets when helpful
#
# Footer rules:
# - Use footers for references or metadata, e.g. Refs: #23, Closes: #45
# - Breaking changes may also be written as:
#   BREAKING CHANGE: describe the incompatible change
#
# Remember:
# - feat and fix map to SemVer intent
# - Use ! or BREAKING CHANGE: for incompatible API changes
# - Keep the first line short and specific
# -----------------------------------------------------------------------
